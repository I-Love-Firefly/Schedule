cmake_minimum_required(VERSION 3.22.1)
project(schedule_ai LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_KLEIDIAI ON CACHE BOOL "" FORCE)

if(NOT DEFINED LLAMA_SRC OR NOT EXISTS "${LLAMA_SRC}/CMakeLists.txt")
    message(FATAL_ERROR "Pass -DLLAMA_SRC=<path to llama.cpp-omni>")
endif()
add_subdirectory("${LLAMA_SRC}" "${CMAKE_BINARY_DIR}/llama")

add_library(schedule_ai SHARED schedule_ai.cpp)
target_include_directories(schedule_ai PRIVATE "${LLAMA_SRC}/include" "${LLAMA_SRC}/ggml/include")
target_link_libraries(schedule_ai PRIVATE llama)
target_compile_options(schedule_ai PRIVATE -fvisibility=hidden -ffunction-sections -fdata-sections)
target_link_options(schedule_ai PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384)
